class DashboardPage extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return BlocBuilder<DashboardBloc, DashboardState>(
      builder: (context, state) {
        return Scaffold(
          body: StreamBuilder<List<Order>>(
            stream: getIt<OrderRepository>().watchTodayOrders(),
            builder: (context, snapshot) {
              return OrderSummaryCard(orders: snapshot.data ?? []);
            },
          ),
        );
      },
    );
  }
}
CREATE TABLE orders (
  id TEXT PRIMARY KEY,               -- Client UUID
  server_order_no TEXT,              -- e.g. ORD-2026-1002
  customer_id TEXT NOT NULL,
  status TEXT NOT NULL,              -- draft, confirmed, syncing, synced, needs_review, failed
  notes TEXT,
  subtotal_paise INTEGER NOT NULL,   -- e.g. 480000 (= ₹4,800.00)
  discount_paise INTEGER DEFAULT 0,  -- Integer paise eliminates float error
  total_paise INTEGER NOT NULL,      -- Net total in paise
  created_at TEXT NOT NULL,
  confirmed_at TEXT,
  sync_status TEXT NOT NULL,         -- pending, synced, failed, needs_review
  sync_error TEXT,
  retry_count INTEGER DEFAULT 0
);

Outbox & Idempotency Specification

  • Local-First: Every order creation writes to orders and outbox in a single SQLite transaction.
  • Idempotency-Key: Generated on the client using idemp_${uuid}. Retries with the same key return 200 OK without creating duplicate orders.
  • Exponential Backoff: Retries interval sequence: 30s ➡️ 1m ➡️ 2m ➡️ 5m ➡️ capped at 15m with jitter.
  • 409 Conflict Handling: When stock is exhausted on server, order moves to needs_review status with line-level conflict indicators. Rep can edit quantity and resubmit.
